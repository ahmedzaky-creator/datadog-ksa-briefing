# 1. Parse JSON payload safely
.parsed, err = parse_json(.message)
if err == null { . = merge(., .parsed) }

# 2. Redact Saudi National ID (10 digits starting with 1)
.message = replace(.message, r'([^\d]|^)[1][0-9]{9}([^\d]|$)', "$1[REDACTED_SA_NID]$2")

# 3. Redact Saudi Resident Iqama (10 digits starting with 2)
.message = replace(.message, r'([^\d]|^)[2][0-9]{9}([^\d]|$)', "$1[REDACTED_IQAMA]$2")

# 4. Redact Saudi IBAN (SA + 22 alphanumeric digits)
.message = replace(.message, r'SA[0-9]{2}[A-Z0-9]{20}', "[REDACTED_SA_IBAN]")

# 5. Redact Credit Card PAN (Luhn validated PANs)
.message = replace(.message, r'\b(?:4[0-9]{12}(?:[0-9]{3})?|5[1-5][0-9]{14})\b', "[REDACTED_PAN]")
